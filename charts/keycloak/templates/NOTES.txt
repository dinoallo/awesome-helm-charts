1. Get the Keycloak admin URL:
{{- if .Values.ingress.enabled }}
{{- range .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}{{ range .paths }}{{ .path }}{{ end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "keycloak.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  You can watch the status by running:
    kubectl get svc --namespace {{ .Release.Namespace }} -w {{ include "keycloak.fullname" . }}

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "keycloak.fullname" . }} --template "{{ "{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}" }}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- end }}

2. Login with username: {{ .Values.auth.adminUser }}
{{- if .Values.auth.existingAdminPasswordSecret }}
   Admin password is stored in the Secret "{{ .Values.auth.existingAdminPasswordSecret }}" (key: admin-password).
   Retrieve it with:
     kubectl get secret --namespace {{ .Release.Namespace }} {{ .Values.auth.existingAdminPasswordSecret }} -o jsonpath="{.data.admin-password}" | base64 -d
{{- else if .Values.auth.adminPassword }}
   Admin password was configured during install. To avoid exposing credentials in logs, consider switching to auth.existingAdminPasswordSecret.
{{- else }}
   Admin password is not configured. Set auth.adminPassword or auth.existingAdminPasswordSecret.
{{- end }}

{{- if ne .Values.database.vendor "dev-file" }}
3. Database: {{ .Values.database.vendor }} on {{ .Values.database.hostname }}:{{ .Values.database.port }}/{{ .Values.database.database }}
{{- else }}
3. **WARNING**: Using the built-in H2 database (dev-file). This is NOT suitable for production.
   Configure a production-grade database (PostgreSQL, MySQL, MariaDB, etc.) via extraEnv.
{{- end }}
